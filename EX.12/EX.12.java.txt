
import java.util.*;

class Course {
private String name;
private int seats;

Course(String name, int seats) {
    this.name = name;
    this.seats = seats;
}

public String getName() {
    return name;
}

public int getSeats() {
    return seats;
}

public boolean register() {
    if (seats > 0) {
        seats--;
        return true;
    }
    return false;
}

}

public class CollegeCourseRegistrationSystem {
public static void main(String[] args) {
Scanner sc = new Scanner(System.in);

    ArrayList<Course> courses = new ArrayList<>();
    ArrayList<String> registered = new ArrayList<>();

    courses.add(new Course("Java", 2));
    courses.add(new Course("Python", 3));
    courses.add(new Course("Database", 2));

    while (true) {
        System.out.println("\n1. View Courses");
        System.out.println("2. Register Course");
        System.out.println("3. View Registered Courses");
        System.out.println("4. Exit");
        System.out.print("Enter choice: ");

        try {
            int choice = Integer.parseInt(sc.nextLine());

            if (choice == 1 || choice == 2) {
                for (int i = 0; i < courses.size(); i++)
                    System.out.println((i + 1) + ". "
                            + courses.get(i).getName()
                            + " - Seats: "
                            + courses.get(i).getSeats());

                if (choice == 2) {
                    System.out.print("Select course number: ");
                    int n = Integer.parseInt(sc.nextLine());

                    if (n < 1 || n > courses.size()) {
                        System.out.println("Invalid course!");
                    } else {
                        Course c = courses.get(n - 1);

                        if (registered.contains(c.getName()))
                            System.out.println("Already registered!");
                        else if (c.register()) {
                            registered.add(c.getName());
                            System.out.println("Registration successful!");
                        } else
                            System.out.println("No seats available!");
                    }
                }
            } else if (choice == 3) {
                System.out.println("Registered Courses: " + registered);
            } else if (choice == 4) {
                System.out.println("Thank you!");
                break;
            } else {
                System.out.println("Invalid choice!");
            }
        } catch (NumberFormatException e) {
            System.out.println("Enter a valid number!");
        }
    }

    sc.close();
}

}
